-- 2. Tables with Constraints
CREATE TABLE Department (
    Dept_ID INT PRIMARY KEY,
    Dept_Name VARCHAR(50),
    Location VARCHAR(50)
);

CREATE TABLE Student (
    Student_ID INT PRIMARY KEY,
    Name VARCHAR(50) NOT NULL,
    Age INT CHECK (Age >= 18),
    Course VARCHAR(50),
    City VARCHAR(50),
    Dept_ID INT,
    Course_ID INT
);

CREATE TABLE Employee (
    Emp_ID INT PRIMARY KEY,
    Emp_Name VARCHAR(50),
    Department VARCHAR(50),
    Salary DECIMAL(10,2),
    Email VARCHAR(100) UNIQUE,
    Manager_ID INT,
    Joining_Date DATE
);

CREATE TABLE Product (
    Product_ID INT PRIMARY KEY,
    Product_Name VARCHAR(50),
    Price DECIMAL(10,2) CHECK (Price > 0),
    Quantity INT
);

CREATE TABLE Customer (
    Customer_ID INT PRIMARY KEY,
    Customer_Name VARCHAR(50),
    Email VARCHAR(100),
    Phone VARCHAR(20)
);

CREATE TABLE Course (
    Course_ID INT PRIMARY KEY,
    Course_Name VARCHAR(50),
    Duration VARCHAR(20),
    Fees DECIMAL(10,2)
);

-- =========================================================================
-- Lab Sheet 3: SQL Functions
-- =========================================================================

-- A. Number Functions
SELECT ABS(-15);
SELECT ROUND(123.456, 2);
SELECT CEIL(4.2), FLOOR(4.8);
SELECT MOD(10, 3);
SELECT Emp_Name, ROUND(Salary, 2) AS Rounded_Salary FROM Employee;

-- B. Aggregate Functions
SELECT COUNT(*) AS Total_Students FROM Student;
SELECT SUM(Salary) AS Total_Salary FROM Employee;
SELECT AVG(Salary) AS Average_Salary FROM Employee;
SELECT MAX(Salary) AS Highest_Salary FROM Employee;
SELECT MIN(Salary) AS Lowest_Salary FROM Employee;

-- C. Character Functions
SELECT UPPER(Name) FROM Student;
SELECT LOWER(Name) FROM Student;
SELECT Name, LENGTH(Name) AS Name_Length FROM Student;
-- Note: CONCAT example assuming First_Name and Last_Name columns exist or demonstrated conceptually
SELECT CONCAT('John', ' ', 'Doe') AS Full_Name;
SELECT SUBSTRING(Name, 1, 3) FROM Student;

-- D. Conversion Functions
SELECT CAST(123 AS CHAR);
SELECT CAST('123' AS SIGNED);
SELECT DATE_FORMAT(NOW(), '%d-%m-%Y');
SELECT CAST('2026-06-06' AS DATE);
SELECT CONVERT('123', SIGNED);

-- E. Date Functions
SELECT CURDATE();
SELECT YEAR(CURDATE()), MONTH(CURDATE()), DAY(CURDATE());
SELECT DATEDIFF('2026-12-31', '2026-01-01');
SELECT DATE_ADD(CURDATE(), INTERVAL 10 DAY);
SELECT Emp_Name, DATE_FORMAT(Joining_Date, '%d/%m/%Y') AS Formatted_Date FROM Employee;


-- =========================================================================
-- Lab Sheet 4: SQL Operators
-- =========================================================================

-- Arithmetic Operators
SELECT 10 + 5;
SELECT 10 - 5;
SELECT 10 * 5;
SELECT 10 / 2;
SELECT Price * Quantity AS Total_Amount FROM Product;

-- Logical Operators
SELECT * FROM Student WHERE Age > 18 AND City = 'Delhi';
SELECT * FROM Student WHERE City = 'Delhi' OR City = 'Noida';
SELECT * FROM Student WHERE NOT (City = 'Delhi');
SELECT * FROM Student WHERE Course = 'Computer Science' AND City = 'Delhi';
SELECT * FROM Employee WHERE Department = 'HR' OR Department = 'IT';

-- Comparison Operators
SELECT * FROM Student WHERE City = 'Delhi';
SELECT * FROM Employee WHERE Salary > 50000;
SELECT * FROM Student WHERE Age < 20;
SELECT * FROM Employee WHERE Salary >= 30000 AND Salary <= 70000;
SELECT * FROM Employee WHERE Department != 'HR';

-- Special Operators
SELECT * FROM Employee WHERE Salary BETWEEN 30000 AND 60000;
SELECT * FROM Student WHERE City IN ('Delhi', 'Noida', 'Meerut');
SELECT * FROM Student WHERE City NOT IN ('Delhi', 'Noida');
SELECT * FROM Student WHERE Name LIKE 'A%';
SELECT * FROM Student WHERE Email IS NULL;

-- Set Operations
SELECT City FROM Student UNION SELECT Location FROM Department;
SELECT City FROM Student UNION ALL SELECT Location FROM Department;
SELECT City FROM Student INTERSECT SELECT Location FROM Department;
SELECT City FROM Student EXCEPT SELECT Location FROM Department;

-- Multiple operators combined
SELECT Emp_Name, Salary, Department FROM Employee WHERE (Salary + 5000) > 50000 AND Department IN ('HR', 'IT') AND Emp_Name LIKE 'J%';


-- =========================================================================
-- Lab Sheet 5: SQL Joins
-- =========================================================================

SELECT * FROM Student INNER JOIN Department ON Student.Dept_ID = Department.Dept_ID;
SELECT Student.Name, Department.Dept_Name FROM Student INNER JOIN Department ON Student.Dept_ID = Department.Dept_ID;
SELECT Employee.Emp_Name, Department.Dept_Name FROM Employee INNER JOIN Department ON Employee.Department = Department.Dept_Name;
SELECT Student.Name, Course.Course_Name, Course.Duration FROM Student INNER JOIN Course ON Student.Course_ID = Course.Course_ID;
SELECT * FROM Student INNER JOIN Department ON Student.Dept_ID = Department.Dept_ID WHERE Department.Dept_Name = 'Computer Science';

SELECT * FROM Student LEFT OUTER JOIN Department ON Student.Dept_ID = Department.Dept_ID;
SELECT Student.Name, Department.Dept_Name FROM Student LEFT JOIN Department ON Student.Dept_ID = Department.Dept_ID;
SELECT * FROM Student RIGHT OUTER JOIN Department ON Student.Dept_ID = Department.Dept_ID;
SELECT Student.Name, Department.Dept_Name FROM Student RIGHT JOIN Department ON Student.Dept_ID = Department.Dept_ID;

SELECT Student.* FROM Student LEFT JOIN Department ON Student.Dept_ID = Department.Dept_ID WHERE Department.Dept_ID IS NULL;
SELECT Department.* FROM Department LEFT JOIN Student ON Department.Dept_ID = Student.Dept_ID WHERE Student.Student_ID IS NULL;

-- FULL OUTER JOIN workaround
SELECT * FROM Student LEFT JOIN Department ON Student.Dept_ID = Department.Dept_ID 
UNION 
SELECT * FROM Student RIGHT JOIN Department ON Student.Dept_ID = Department.Dept_ID;

SELECT * FROM Student CROSS JOIN Course;
SELECT COUNT(*) FROM Student CROSS JOIN Course;

SELECT * FROM Student NATURAL JOIN Department;

-- SELF JOIN
SELECT e.Emp_Name AS Employee, m.Emp_Name AS Manager FROM Employee e LEFT JOIN Employee m ON e.Manager_ID = m.Emp_ID;
SELECT e.Emp_Name AS Employee, m.Emp_Name AS Manager FROM Employee e INNER JOIN Employee m ON e.Manager_ID = m.Emp_ID;

-- Multi-table joins
SELECT Student.Name, Course.Course_Name, Department.Dept_Name 
FROM Student 
JOIN Department ON Student.Dept_ID = Department.Dept_ID 
JOIN Course ON Student.Course_ID = Course.Course_ID;

SELECT s.Name, c.Course_Name, d.Dept_Name 
FROM Student s 
JOIN Department d ON s.Dept_ID = d.Dept_ID 
JOIN Course c ON s.Course_ID = c.Course_ID;

-- Subqueries & Group By
SELECT e.* FROM Employee e WHERE e.Salary > (SELECT AVG(sub.Salary) FROM Employee sub WHERE sub.Department = e.Department);

SELECT d.Dept_Name FROM Department d JOIN Student s ON d.Dept_ID = s.Dept_ID GROUP BY d.Dept_ID, d.Dept_Name ORDER BY COUNT(s.Student_ID) DESC LIMIT 1;

SELECT d.Dept_Name, COUNT(s.Student_ID) AS Total_Students FROM Department d LEFT JOIN Student s ON d.Dept_ID = s.Dept_ID GROUP BY d.Dept_ID, d.Dept_Name;

SELECT c.Course_Name, COUNT(s.Student_ID) AS Enrolled_Students FROM Course c LEFT JOIN Student s ON c.Course_ID = s.Course_ID GROUP BY c.Course_ID, c.Course_Name;

SELECT d.Dept_Name, c.Course_Name, COUNT(s.Student_ID) AS Total_Students 
FROM Department d 
LEFT JOIN Student s ON d.Dept_ID = s.Dept_ID 
LEFT JOIN Course c ON s.Course_ID = c.Course_ID 
GROUP BY d.Dept_Name, c.Course_Name;
